  const relatedScenes = getFieldValue(sheet, "Related Scenes");
  const relatedDocs = getFieldValue(sheet, "Related Documents");
  const dependencies = getFieldValue(sheet, "Dependencies");
  const confidentiality = getFieldValue(sheet, "Confidentiality");
  const notes = getFieldValue(sheet, "Notes");
  
  // Generate document ID
  const docId = generateDocumentId(department, documentType, topic, version);
  
  // Construct metadata block
  let metadata = "---PRODUCTION METADATA---\n";
  metadata += `PRODUCTION: ${production}\n`;
  metadata += `DOCUMENT ID: ${docId}\n`;
  metadata += `DEPARTMENT: ${department}\n`;
  metadata += `CREATOR: ${creator}\n`;
  metadata += `POSITION: ${position}\n`;
  metadata += `CREATION DATE: ${formatDate(creationDate)}\n`;
  metadata += `LAST MODIFIED: ${formatDate(modifiedDate)}\n`;
  metadata += `VERSION: ${version}\n`;
  metadata += `STATUS: ${status}\n`;
  metadata += `CERTAINTY LEVEL: ${certainty}\n`;
  metadata += `PRODUCTION PHASE: ${phase}\n`;
  metadata += `DOCUMENT TYPE: ${documentType}\n`;
  
  // Add optional fields if they have values
  if (relatedScenes) metadata += `RELATED SCENES: ${relatedScenes}\n`;
  if (relatedDocs) metadata += `RELATED DOCUMENTS: ${relatedDocs}\n`;
  if (dependencies) metadata += `DEPENDENCIES: ${dependencies}\n`;
  if (confidentiality) metadata += `CONFIDENTIALITY: ${confidentiality}\n`;
  if (notes) metadata += `NOTES: ${notes}\n`;
  
  metadata += "---END METADATA---";
  
  // Update the preview area in the sheet
  updateMetadataPreview(sheet, metadata);
  
  // Generate filename
  const filename = generateFilename(production, department, documentType, topic, creator, status, version);
  updateFilenamePreview(sheet, filename);
  
  return {metadata: metadata, filename: filename};
}

  if (!date) return "";
  
  // If it's already a string, try to parse it

    date = new Date(date);

/**
 * Formats date objects to YYYY-MM-DD
 */
function formatDate(date) {
  if (!date) return "";
  
  // If it's already a string, try to parse it
  if (typeof date === 'string') {
    date = new Date(date);
  }
  
  if (date instanceof Date) {
    return Utilities.formatDate(date, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  
  return date;
}

/**
 * Helper function to create a dropdown validation from sheet data
 */
function createDropdownFromSheet(targetCell, sheetName, valueColumn, includeColumn, includeValue) {
  const sheet = SS.getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  
  // Filter values that have "X" in the include column
  const values = [];
  for (let i = 1; i < data.length; i++) {
    const colIndex = getColumnIndex(valueColumn);
    const includeColIndex = getColumnIndex(includeColumn);
    
    if (data[i][includeColIndex] === includeValue) {
      values.push(data[i][colIndex]);
    }
  }
  
  // Create the validation rule
  if (values.length > 0) {
    const rule = SpreadsheetApp.newDataValidation().requireValueInList(values).build();
    targetCell.setDataValidation(rule);
  }
}

/**
 * Converts column letter to index (0-based)
 */
function getColumnIndex(column) {
  if (typeof column === 'number') return column;
  
  let result = 0;
  for (let i = 0; i < column.length; i++) {
    result = result * 26 + (column.charCodeAt(i) - 64);
  }
  return result - 1; // Convert to 0-based index
}
